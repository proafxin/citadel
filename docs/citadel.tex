	% SLIDE 6: WHY CITADEL IS DIFFERENT (ARCHITECTURE SIMPLIFIED)
	\begin{frame}
		\frametitle{Why Citadel is Different}
		We replaced traditional RAG text-splitting with a three-layer production architecture:
		\vfill
		\begin{description}
			\item[\textbf{1. Intelligent Ingestion:}] Natively preserves multi-dimensional tables, layouts, formulas, and visual relationships instead of flattening documents into generic text.
			\item[\textbf{2. Structured Knowledge Engine:}] Creates searchable, interconnected data objects instead of isolated, fragmented text chunks.
			\item[\textbf{3. Evidence-First Retrieval:}] Answers are synthesized strictly from verified evidence clusters, preventing standard vector embedding hallucinations.
		\end{description}
	\end{frame}

	\begin{frame}
		\frametitle{Private by Design}
		\begin{itemize}
			\item \textbf{Zero Training Guardrails:} Customer document assets are never utilized to train, fine-tune, or modify underlying model weights.
			\item \textbf{Bring Your Own LLM:} Connects ephemerally to your existing enterprise infrastructure (OpenAI API, Azure, or entirely local weights).
			\item \textbf{Walled Sandbox Indexing:} Files are processed on-the-fly into localized layout structures with absolute data isolation.
			\item \textbf{Cloud Optional:} Fully operational inside on-premise air-gapped secure facilities or private clouds.
		\end{itemize}
