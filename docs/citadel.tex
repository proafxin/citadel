	% ---------------- SLIDE 1 ----------------
	\begin{frame}{The Problem}
		Companies are drowning in documents:
		\begin{itemize}
			\item PDFs, Excel files, contracts, reports, scanned records
		\end{itemize}

		\vspace{0.4cm}

		But they are not truly usable.

		\vspace{0.3cm}

		\textbf{When someone asks:}
		\begin{quote}
			What changed in revenue across all reports since 2019?
		\end{quote}

		The answer is still:
		\begin{itemize}
			\item manual work
			\item spreadsheets rebuilt from scratch
			\item human interpretation
		\end{itemize}

		\vspace{0.3cm}
		\textbf{Enterprise knowledge exists — but it is not computable.}
	\end{frame}

	% ---------------- SLIDE 2 ----------------
	\begin{frame}{Why This Is Broken}
		Current AI systems try to “read” documents:

		\begin{itemize}
			\item OCR extracts text
			\item LLMs summarize it
			\item search systems embed it
		\end{itemize}

		\vspace{0.3cm}

		But one critical thing is lost:

		\vspace{0.2cm}
		\textbf{structure}

		\vspace{0.3cm}

		Tables become text. Layout becomes noise. Relationships disappear.

		\vspace{0.3cm}

		So the AI is forced to guess.

		\vspace{0.2cm}
		\textbf{Guessing is unacceptable in financial, legal, and regulated workflows.}
	\end{frame}

	% ---------------- SLIDE 3 ----------------
	\begin{frame}{The Real Bottleneck}
		Answering important questions today requires:

		\begin{itemize}
			\item manual document reconstruction
			\item repeated data cleaning
			\item multiple teams redoing the same work
		\end{itemize}

		\vspace{0.3cm}

		\textbf{Result:}
		\begin{itemize}
			\item days to answer simple questions
			\item inconsistent outputs
			\item high risk of error
		\end{itemize}

		\vspace{0.3cm}

		\textbf{The bottleneck is reconstruction, not access.}
	\end{frame}

	% ---------------- SLIDE 4 ----------------
	\begin{frame}{The Insight}
		We don’t need better document summaries.

		\vspace{0.4cm}

		We need something different:

		\vspace{0.3cm}

		\begin{center}
			\Large
			\textbf{Documents should behave like databases.}
		\end{center}

		\vspace{0.3cm}

		Not text. Not embeddings. Not summaries.

		\vspace{0.2cm}

		But structured, queryable data.
	\end{frame}

	% ---------------- SLIDE 5 ----------------
	\begin{frame}{The Citadel Solution}
		Citadel converts any document into structured data:

		\begin{itemize}
			\item PDFs
			\item Excel files
			\item Word documents
			\item scanned images
			\item CSV / JSON
		\end{itemize}

		\vspace{0.3cm}

		Into:

		\vspace{0.2cm}

		\begin{center}
			\textbf{a structured, queryable system of record}
		\end{center}
	\end{frame}

	% ---------------- SLIDE 6 ----------------
	\begin{frame}{How It Works}
		\begin{enumerate}
			\item Ingest documents
			\item Extract structure (tables, layout, text)
			\item Store as relational data
			\item Let AI generate queries
			\item Database returns exact answers
		\end{enumerate}

		\vspace{0.4cm}

		\begin{center}
			\textbf{AI writes queries. Database computes truth.}
		\end{center}
	\end{frame}

	% ---------------- SLIDE 7 ----------------
	\begin{frame}{Key Difference}
		Current systems:
		\begin{itemize}
			\item flatten everything into text
			\item lose structure permanently
			\item rely on AI to guess meaning
		\end{itemize}

		\vspace{0.4cm}

		Citadel:
		\begin{itemize}
			\item preserves structure at ingestion
			\item stores real relational data
			\item never lets AI invent values
		\end{itemize}

		\vspace{0.3cm}

		\textbf{We remove ambiguity at the source.}
	\end{frame}
