	% ---------------- SLIDE 1 ----------------
	\begin{frame}{Enterprise Data is Not Usable at Query Time}
		Companies already have the data:

		\begin{itemize}
			\item PDFs, Excel models, contracts, reports
		\end{itemize}

		\vspace{0.4cm}

		But when leaders ask real questions:

		\begin{quote}
			What changed in revenue across all reports since 2019?
		\end{quote}

		the answer is still:

		\begin{itemize}
			\item manual reconstruction
			\item spreadsheet rebuilding
			\item human interpretation
		\end{itemize}

		\vspace{0.3cm}

		\textbf{Enterprise knowledge exists — but it is not computable.}
	\end{frame}

	% ---------------- SLIDE 2 ----------------
	\begin{frame}{Why Existing AI Systems Fail}
		Modern AI pipelines:

		\begin{itemize}
			\item OCR → text
			\item chunking → embeddings
			\item LLM → reasoning
		\end{itemize}

		\vspace{0.4cm}

		But one thing is lost permanently:

		\vspace{0.2cm}
		\textbf{structure}

		\vspace{0.3cm}

		Tables become text. Layout disappears. Relationships break.

		\vspace{0.3cm}

		So AI is forced to guess.

		\vspace{0.2cm}

		\textbf{Guessing breaks financial, legal, and audit-grade workflows.}
	\end{frame}

	% ---------------- SLIDE 3 ----------------
	\begin{frame}{The Real Problem}
		Enterprise workflows today are not search problems.

		They are reconstruction problems:

		\begin{itemize}
			\item re-reading documents
			\item rebuilding tables manually
			\item re-validating numbers repeatedly
		\end{itemize}

		\vspace{0.4cm}

		\textbf{Result:}
		\begin{itemize}
			\item days to answer simple questions
			\item inconsistent outputs across teams
			\item high operational risk
		\end{itemize}

		\vspace{0.3cm}

		\textbf{The bottleneck is reconstruction, not access.}
	\end{frame}

	% ---------------- SLIDE 4 ----------------
	\begin{frame}{Insight}
		We don’t need better document understanding.

		\vspace{0.4cm}

		We need a new primitive:

		\vspace{0.3cm}

		\begin{center}
			\Large
			\textbf{Documents should behave like databases}
		\end{center}

		\vspace{0.3cm}

		Not text. Not embeddings. Not summaries.

		\vspace{0.2cm}

		But structured, queryable data.
	\end{frame}

	% ---------------- SLIDE 5 ----------------
	\begin{frame}{The Citadel Approach}
		Citadel converts documents into structured systems of record.

		\vspace{0.3cm}

		We ingest:

		\begin{itemize}
			\item PDFs
			\item Excel files
			\item Word documents
			\item scans, CSV, JSON
		\end{itemize}

		\vspace{0.3cm}

		And compile them into:

		\begin{center}
			\textbf{a relational, queryable enterprise knowledge layer}
		\end{center}
	\end{frame}

	% ---------------- SLIDE 6 ----------------
	\begin{frame}{How It Works}
		\begin{enumerate}
			\item Ingest any document format
			\item Extract full structure (tables, layout, text)
			\item Store as relational data (Postgres)
			\item AI generates SQL queries
			\item Database returns exact answers
		\end{enumerate}

		\vspace{0.4cm}

		\begin{center}
			\Large
			\textbf{AI writes queries. Database computes truth.}
		\end{center}
	\end{frame}
